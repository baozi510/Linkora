cmake_minimum_required(VERSION 3.16)
project(linkora_ffmpeg LANGUAGES C CXX)

if(NOT OHOS_ARCH STREQUAL "arm64-v8a" AND NOT OHOS_ARCH STREQUAL "x86_64")
  message(FATAL_ERROR "linkora_ffmpeg supports OHOS arm64-v8a and x86_64 only; got ${OHOS_ARCH}")
endif()
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(FFMPEG_PREBUILT_ROOT "${REPO_ROOT}/third_party/ffmpeg/prebuilt" CACHE PATH "Pinned FFmpeg bootstrap output")
set(PREFIX "${FFMPEG_PREBUILT_ROOT}/${OHOS_ARCH}")
if(NOT EXISTS "${PREFIX}/linkora-build-manifest.txt")
  message(FATAL_ERROR "Missing FFmpeg ${OHOS_ARCH} prebuilt at ${PREFIX}. Run scripts/ffmpeg/fetch-source.ps1 then scripts/ffmpeg/build-harmony.sh ${OHOS_ARCH} <HarmonyOS-native-root>; do not use host headers/libs.")
endif()
file(READ "${PREFIX}/linkora-build-manifest.txt" MANIFEST)
if(NOT MANIFEST MATCHES "ffmpeg_commit=1041abdc962f4cc4f394aa8de9dc5236c0c3b9e7" OR
   NOT MANIFEST MATCHES "abi=${OHOS_ARCH}")
  message(FATAL_ERROR "FFmpeg prebuilt manifest must match the exact 8.1.3 commit and ${OHOS_ARCH} ABI")
endif()
foreach(HEADER libavformat/avformat.h libavcodec/avcodec.h libavutil/avutil.h libswscale/swscale.h)
  if(NOT EXISTS "${PREFIX}/include/${HEADER}")
    message(FATAL_ERROR "Missing FFmpeg ${OHOS_ARCH} header ${HEADER}; rebuild the pinned bootstrap")
  endif()
endforeach()
get_filename_component(LLVM_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
set(READELF "${LLVM_BIN}/llvm-readelf")
if(EXISTS "${READELF}.exe")
  set(READELF "${READELF}.exe")
endif()
set(EXPECTED_MACHINE "AArch64")
if(OHOS_ARCH STREQUAL "x86_64")
  set(EXPECTED_MACHINE "Advanced Micro Devices X86-64")
endif()
foreach(LIB avformat avcodec avutil swscale)
  set(ARCHIVE "${PREFIX}/lib/lib${LIB}.a")
  if(NOT EXISTS "${ARCHIVE}")
    message(FATAL_ERROR "Missing FFmpeg ${OHOS_ARCH} lib${LIB}.a; rebuild the pinned bootstrap")
  endif()
  execute_process(COMMAND "${READELF}" -h "${ARCHIVE}" RESULT_VARIABLE STATUS OUTPUT_VARIABLE ELF ERROR_VARIABLE ELF_ERROR)
  string(REGEX MATCHALL "Machine:[^\n\r]+" MACHINES "${ELF}")
  if(NOT STATUS EQUAL 0 OR NOT MACHINES)
    message(FATAL_ERROR "Cannot audit FFmpeg archive ${ARCHIVE}: ${ELF_ERROR}")
  endif()
  foreach(MACHINE ${MACHINES})
    if(NOT MACHINE MATCHES "${EXPECTED_MACHINE}")
      message(FATAL_ERROR "Wrong FFmpeg archive ABI: ${ARCHIVE} ${MACHINE}; expected ${EXPECTED_MACHINE}")
    endif()
  endforeach()
  add_library(ffmpeg_${LIB} STATIC IMPORTED)
  set_target_properties(ffmpeg_${LIB} PROPERTIES IMPORTED_LOCATION "${ARCHIVE}")
endforeach()

add_library(linkora_ffmpeg SHARED analysis.cpp napi_init.cpp)
target_compile_features(linkora_ffmpeg PRIVATE cxx_std_17)
target_include_directories(linkora_ffmpeg PRIVATE "${PREFIX}/include")
# Bind this module to its own pinned static FFmpeg, never libmpv's internal symbols.
target_link_options(linkora_ffmpeg PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,--no-undefined")
target_link_libraries(linkora_ffmpeg PRIVATE "-Wl,--start-group" ffmpeg_avformat ffmpeg_avcodec
  ffmpeg_avutil ffmpeg_swscale "-Wl,--end-group" libace_napi.z.so m)
