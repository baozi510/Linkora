import re
LABEL='udid of current device is :'
ID_PATTERN=re.compile(r'[0-9A-Fa-f]{64}')
def parse(stdout):
    text=stdout.decode('utf-8',errors='strict')
    lines=[line.strip() for line in text.splitlines() if line.strip()]
    if len(lines)!=2 or lines[0]!=LABEL or not ID_PATTERN.fullmatch(lines[1]):
        raise ValueError('Expected exact observed two-line labeled 64-hex identifier output')
    return lines[1]
