11: /** Composition supplies one shared proxy. This class contains no protocol selection. */
12: export class HarmonyAnalysisInputs implements AnalysisInputPorts {
13:   private readonly store: NetworkServerStore;
14:   private readonly proxy: NetworkFileProxy;
15:   constructor(context: common.UIAbilityContext, proxy: NetworkFileProxy) {
16:     this.store = new NetworkServerStore(context); this.proxy = proxy;
17:   }
18:   async openRemote(source: MediaSource, operation: AnalysisOperation): Promise<RandomAccessSource> {
19:     const servers = await this.store.load(); operation.check();
20:     const server = servers.find((entry) => entry.id === NetworkMediaSourceFactory.serverId(source));
21:     if (server === undefined) throw new AnalysisError(AnalysisErrorCode.RESOLVE_FAILED);
22:     const directory = new NetworkDirectoryService(server);
23:     operation.onCancel(() => directory.cancel());
24:     // MediaSource.fingerprint is a media/cache identity, not an SFTP host-key fingerprint.
25:     // Pass an empty per-open override so the SFTP provider keeps ownership of persisted trust.
26:     let setupSettled: Promise<void> = Promise.resolve();
27:     try {
28:       return await directory.openSource(source.locator, '',
29:         (settled: Promise<void>) => { setupSettled = settled; });
30:     } catch (error) {
31:       // Native setup may time out before a late handle finishes opening/closing. Preserve the
32:       // pre-Phase-3 serialization guarantee so the next loader job does not overlap that cleanup.
33:       await setupSettled.catch(() => {});
34:       throw error;
35:     }
36:   }
37:   async openHttp(source: MediaSource, _operation: AnalysisOperation): Promise<RandomAccessSource> {
38:     // The resolver closes a late HEAD result if cancellation happened while opening.
39:     return new RemoteReadSessionAdapter(source.sourceId, await HttpRemoteReadSession.open(source.locator, {}));
40:   }
41:   async register(source: RandomAccessSource): Promise<AnalysisProxyRegistration> {
42:     const lease = await this.proxy.buildSourceUrl(source, 'application/octet-stream');
43:     return new AnalysisProxyRegistration(lease.url, () => lease.release());
44:   }
