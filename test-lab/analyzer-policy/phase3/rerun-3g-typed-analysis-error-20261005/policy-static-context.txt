88:     Promise<NetworkMediaAnalysisResult> {

28:     this.system = system;
29:     this.ffmpeg = ffmpeg;
30:     this.policy = policy;

32: 
33:   name(): string {
34:     return 'policy';
35:   }
36: 
37:   async canProbe(source: MediaSource): Promise<boolean> {
38:     if (this.closed) return false;
39:     const order: ProductionAnalysisEngine[] =
40:       this.policy.probeOrder(source, ProbeRequirement.LIST);
41:     for (const engine of order) {
42:       if (await this.adapter(engine).canProbe(source)) return true;
43:     }
44:     return false;
45:   }
46: 
47:   async probe(source: MediaSource, requirement: ProbeRequirement): Promise<ProbeResult> {
48:     if (this.closed) {
49:       return new ProbeResult(null, ProbeCompleteness.UNAVAILABLE, [],
50:         new MediaProbeDiagnostics('policy'), AnalysisErrorCode.CLOSED);
51:     }
52: 
53:     const generation: number = this.cancellationGeneration;
54:     const order: ProductionAnalysisEngine[] = this.policy.probeOrder(source, requirement);
55:     let best: ProbeResult | null = null;
56: 
57:     for (let index: number = 0; index < order.length; index++) {
58:       if (generation !== this.cancellationGeneration || this.closed) {
59:         return new ProbeResult(null, ProbeCompleteness.UNAVAILABLE, [],
60:           new MediaProbeDiagnostics('policy'), AnalysisErrorCode.CANCELLED);
61:       }
62:       const adapter: IMediaProbe = this.adapter(order[index]);
63:       if (!await adapter.canProbe(source)) continue;
64: 
65:       const result: ProbeResult = await adapter.probe(source, requirement);
66:       if (generation !== this.cancellationGeneration || this.closed) {
67:         return new ProbeResult(null, ProbeCompleteness.UNAVAILABLE, [],
68:           new MediaProbeDiagnostics('policy'), AnalysisErrorCode.CANCELLED);
69:       }
70:       if (result.completeness === ProbeCompleteness.COMPLETE) return result;
71: 
72:       if (result.isUsable()) {

74:         // FFmpeg is already the richer primary for DETAIL/ADVANCED. A System
75:         // fallback cannot enrich a usable partial FFmpeg result without merging.
76:         if (order[index] === ProductionAnalysisEngine.FFMPEG &&
77:           requirement !== ProbeRequirement.LIST) {
78:           return result;
79:         }
80:       }
81:     }
82: 
83:     return best ?? new ProbeResult(null, ProbeCompleteness.UNAVAILABLE, [],
84:       new MediaProbeDiagnostics('policy'), AnalysisErrorCode.UNSUPPORTED);
85:   }
86: 
87:   cancel(): void {
88:     this.cancellationGeneration++;
89:     this.system.cancel();
90:     this.ffmpeg.cancel();
91:   }

30:       [ProductionAnalysisEngine.SYSTEM] :
31:       [ProductionAnalysisEngine.FFMPEG, ProductionAnalysisEngine.SYSTEM];
32:   }

123:   private friendlyError(error: Object): string {
124:     const raw: string = error instanceof Error ? error.message : `${error}`;
125:     const normalized: string = raw.toLowerCase();
126:     if (normalized.includes('hostkey') || normalized.includes('指纹')) {
127:       return raw.replace(/^SFTP_HOSTKEY:\s*/, '');
128:     }
129:     if (normalized.includes('auth') || normalized.includes('permission') ||
130:       normalized.includes('账号') || normalized.includes('密码')) {
131:       return '账号或密码不正确，或者当前账号没有访问权限。';
132:     }
133:     if (normalized.includes('resolve')) {
134:       return '无法解析服务器地址，请检查主机名。';
135:     }
136:     if (normalized.includes('timeout') || normalized.includes('超时')) {
137:       return '连接超时，请检查服务器地址、端口和网络。';
138:     }
139:     if (normalized.includes('refused') || normalized.includes('connect')) {
140:       return '无法连接 SFTP 服务器，请检查地址和端口。';
141:     }
142:     if (normalized.includes('local_write')) {
143:       return '无法创建播放缓存，请检查设备存储空间。';
144:     }
145:     if (normalized.includes('no such') || normalized.includes('目录')) {
146:       return '远程目录不存在，请检查路径。';
147:     }
148:     return `SFTP 访问失败：${raw.replace(/^Error:\s*/, '')}`;
149:   }

151:   private static normalizeAbsolute(path: string): string {
152:     const value: string = path.trim();
153:     if (value.length === 0 || value === '/') {
154:       return '/';
155:     }
156:     return `/${value.replace(/^\/+|\/+$/g, '')}`;
157:   }
